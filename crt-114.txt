def binary_search(arr,target):
    left=0
    right=len(arr)-1
    while left<right:
        mid=(left+right)//2
        if target==arr[mid]:return target
        elif target<arr[mid]:left=mid+1
        else                :right-mid-1
    return -1
arr=[24,67,90,98,99,102]
target=98
idx=binary_search(arr,target)
print(f"binary_search is: {idx}")
