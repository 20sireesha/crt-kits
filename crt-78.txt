n=7
Revenue=[]
for i in range(n):
    ele=int(input(f"Enter the Revenue for day {i+1}:"))
    Revenue.append(ele)
print(f"Total Revenue {sum(Revenue)} | Best Day:{max(Revenue)} | Worst Day:{min(Revenue)}")

#using split
Revenue=list(map(int,input("Enter the Revenuefor 7 days:").split()))
print(f"Total Revenue {sum(Revenue)} | Best Day:{max(Revenue)} | Worst Day:{min(Revenue)}")
