#write a python program to read the month number as input from the user and print the respective number of days of particular number of month
month=int(input("Enter month number:"))
if(month in[1,3,5,7,8,10,12]):
    print("31 days")
elif(month in[4,6,9,11]):
    print("30 days")
elif(month==2):
    print("28 or 29 days")
else:
    ("Invalid month number")
