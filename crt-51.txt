pin=int(input("Enter the pin:"))
acc_bal=0
if pin==1234:
    print("Welcome to Bank")
    while True:
        print("1.Deposite ")
        print("2.Withdrawl")
        print("3.Balane Inquiry")
        print("4.Exit")
 
        choice=int(input("Enter the choice:"))
        print("\n")
        if(choice==1):
            amount=int(input("Enter the amount to deposite:"))
            acc_bal=acc_bal+amount
            print(f"Dear customer your account xxxxxxxxxxxxx1234 is credited with {amount}")
        elif(choice==2):
            amount=int(input("Enter the amount to withdraw:"))
            if(amount<acc_bal):
                print(f"Dear customer your account xxxxxxxxxxxxx1234 is debited with {amount}")
                acc_bal=acc_bal-amount
            else:
                print(f"Insufficient amount")
        elif(choice==3):
            print(f"Dear customer your account xxxxxxxxxxxxx1234 has {acc_bal}")
        else:
            ("Thank you.....")
            break
